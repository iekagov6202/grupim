Private Function KeyExists(coll As Collection, key As String) As Boolean
    Dim tmp As Variant
    On Error Resume Next
    tmp = coll.Item(key)
    KeyExists = (Err.Number = 0)
    Err.Clear
    On Error GoTo 0
End Function
